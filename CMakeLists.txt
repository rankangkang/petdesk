cmake_minimum_required(VERSION 3.21)
if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum supported macOS")
endif()
project(PetDesk VERSION 0.4.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(APPLE)
    # Qt 6.9.0 has QTBUG-134784 in Cocoa accessibility cache ownership.
    find_package(Qt6 6.9.1 REQUIRED COMPONENTS Core Gui Widgets)
else()
    find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets)
endif()
add_executable(petdesk_icon_generator EXCLUDE_FROM_ALL scripts/generate-app-icon.cpp)
target_link_libraries(petdesk_icon_generator PRIVATE Qt6::Gui)

add_library(petformat STATIC src/petformat.cpp)
target_include_directories(petformat PUBLIC src)
target_link_libraries(petformat PUBLIC Qt6::Core Qt6::Gui)
if(MSVC)
    target_compile_options(petformat PRIVATE /W4 /utf-8)
else()
    target_compile_options(petformat PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(APPLE)
    enable_language(OBJCXX)
    set(PETDESK_STATUS_SOURCE src/statusentry-mac.mm)
    set_source_files_properties(src/macos-reopen.mm src/statusentry-mac.mm tests/statusentry_mac_tests.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
else()
    set(PETDESK_STATUS_SOURCE src/statusentry-qt.cpp)
endif()
qt_add_executable(PetDesk MACOSX_BUNDLE WIN32
    src/main.cpp src/behavior.cpp src/petwindow.cpp src/controller.cpp
    src/managerwindow.cpp src/petlist.cpp src/uistyle.cpp src/preferences.cpp src/activation.cpp ${PETDESK_STATUS_SOURCE})
if(APPLE)
    target_sources(PetDesk PRIVATE src/macos-reopen.mm)
    target_link_libraries(PetDesk PRIVATE "-framework Cocoa")
endif()
target_link_libraries(PetDesk PRIVATE petformat Qt6::Widgets)
target_compile_definitions(PetDesk PRIVATE PETDESK_VERSION="${PROJECT_VERSION}")
qt_add_resources(PetDesk petdesk_resources PREFIX "/" FILES assets/PetDesk.png assets/PetDesk-status.png)
if(MSVC)
    target_compile_options(PetDesk PRIVATE /W4 /utf-8)
else()
    target_compile_options(PetDesk PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_target_properties(PetDesk PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER dev.cmkk.petdesk
    MACOSX_BUNDLE_BUNDLE_NAME PetDesk
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION}
    MACOSX_BUNDLE_BUNDLE_VERSION ${PROJECT_VERSION}
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/Info.plist.in")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/assets/PetDesk.icns")
    target_sources(PetDesk PRIVATE assets/PetDesk.icns)
    set_source_files_properties(assets/PetDesk.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
if(WIN32)
    enable_language(RC)
    configure_file(packaging/PetDesk.rc.in "${CMAKE_CURRENT_BINARY_DIR}/PetDesk.rc" @ONLY)
    target_sources(PetDesk PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/PetDesk.rc")
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(importer_tests tests/importer_tests.cpp)
    target_link_libraries(importer_tests PRIVATE petformat Qt6::Test)
    add_test(NAME importer COMMAND importer_tests)
    add_executable(behavior_tests tests/behavior_tests.cpp src/behavior.cpp)
    target_link_libraries(behavior_tests PRIVATE petformat Qt6::Test)
    add_test(NAME behavior COMMAND behavior_tests)
    add_executable(window_tests tests/window_tests.cpp src/behavior.cpp src/petwindow.cpp)
    target_link_libraries(window_tests PRIVATE petformat Qt6::Widgets Qt6::Test)
    add_test(NAME window_lifecycle COMMAND window_tests)
    set_tests_properties(window_lifecycle PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(configuration_tests tests/configuration_tests.cpp src/behavior.cpp
        src/petwindow.cpp src/preferences.cpp src/managerwindow.cpp src/petlist.cpp src/uistyle.cpp src/controller.cpp src/activation.cpp ${PETDESK_STATUS_SOURCE})
    target_link_libraries(configuration_tests PRIVATE petformat Qt6::Widgets Qt6::Test)
    if(APPLE)
        target_link_libraries(configuration_tests PRIVATE "-framework Cocoa")
    endif()
    add_test(NAME configuration COMMAND configuration_tests)
    set_tests_properties(configuration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(statusentry_qt_tests tests/statusentry_qt_tests.cpp src/statusentry-qt.cpp)
    target_include_directories(statusentry_qt_tests PRIVATE src)
    target_link_libraries(statusentry_qt_tests PRIVATE Qt6::Widgets Qt6::Test)
    qt_add_resources(statusentry_qt_tests statusentry_qt_resources PREFIX "/" FILES assets/PetDesk-status.png)
    add_test(NAME statusentry_qt COMMAND statusentry_qt_tests)
    set_tests_properties(statusentry_qt PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(APPLE)
        add_executable(statusentry_mac_tests tests/statusentry_mac_tests.mm src/statusentry-mac.mm)
        target_include_directories(statusentry_mac_tests PRIVATE src)
        target_link_libraries(statusentry_mac_tests PRIVATE Qt6::Widgets Qt6::Test "-framework Cocoa")
        qt_add_resources(statusentry_mac_tests statusentry_mac_resources PREFIX "/" FILES assets/PetDesk-status.png)
        add_test(NAME statusentry_mac COMMAND statusentry_mac_tests)
        set_tests_properties(statusentry_mac PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    endif()
endif()
